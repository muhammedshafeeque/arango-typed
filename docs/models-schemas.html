Models & Schemas

Define schemas using Mongoose-like shorthand and options. Common types: String, Number, Boolean, Date, Array, Object.

const UserSchema = new Schema({
  name: String,
  email: { type: String, required: true, unique: true },
  age: { type: Number, min: 0 },
  tags: { type: Array, of: String, default: [] }
});

const User = model('users', UserSchema);

Validation

const Post = new Schema({
  title: { type: String, required: true },
  slug: { type: String, validate: { validator: v => /\w+/.test(v), message: 'Invalid slug' } }
});

Hooks

UserSchema.pre('save', function(){ this.updatedAt = new Date(); });
UserSchema.post('save', function(){ console.log('saved', this._id); });
